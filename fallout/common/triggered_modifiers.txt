# Triggered modifiers are here.
# these are checked for each countries once/month and then applied.
#
# Effects are fully scriptable here.
world_bank = {
	trigger = {
		has_country_flag = is_bank
	}
	max_tariff = -1.5
	min_tariff = -2
}

ai_diplo = {
	trigger = {
		ai = yes
	}
	diplomatic_points_modifier = -100
}

player_diplo = {
	trigger = {
		ai = no
	}
	diplomatic_points_modifier = 1.0
}

player_skill_issue = {
	trigger = {
		ai = yes
		is_vassal = no
		NOT = {has_country_flag = ces_side}
		NOT = {has_country_flag = nevada_person}
		OR = {
			NOT = {has_country_flag = ncr_side}
			has_country_flag = is_handogs
		}
		NOT = {has_country_flag = is_baja}
	}
	mobilisation_size = -0.03
}
